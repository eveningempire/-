def fuse_rule_scores(rule_scores, weights=None, confidence_scores=None):
    """铻嶅悎澶氫釜瑙勫垯鐨勫垎鏁?""
    if not rule_scores:
        return 0.0
    
    if len(rule_scores) == 1:
        return rule_scores[0]
    
    # 浣跨敤鎸囨暟鍔犳潈骞冲潎锛岄伩鍏嶅紓甯稿€煎奖鍝?
    scores = np.array(rule_scores, dtype=np.float32)
    
    if weights is None:
        weights = np.ones(len(scores))
    
    if confidence_scores is None:
        confidence_scores = np.ones(len(scores))
    
    # 璁＄畻涓庝腑浣嶆暟鐨勫亸宸?
    median_score = np.median(scores)
    deviations = np.abs(scores - median_score)
    
    # 鎸囨暟鏉冮噸锛氬亸宸秺灏忥紝鏉冮噸瓒婂ぇ
    max_deviation = deviations.max()
    if max_deviation == 0:
        return float(median_score)
    
    exp_weights = np.exp(-deviations / max_deviation)
    final_weights = exp_weights * weights * confidence_scores
    
    # 鍔犳潈骞冲潎
    fused_score = np.sum(scores * final_weights) / np.sum(final_weights)
    return float(fused_score)
```

class RuleFusionEngine:
    """瑙勫垯铻嶅悎寮曟搸"""
    
    def __init__(self):
        self.fusion_strategies = {
            'weighted_average': self._weighted_average_fusion,
            'exponential_weighted': self._exponential_weighted_fusion,
            'max_fusion': self._max_fusion,
            'min_fusion': self._min_fusion,
        }
    
    def fuse_rules_by_component(self, rule_results: List[Dict[str, Any]], 
                               strategy: str = 'exponential_weighted') -> Dict[str, float]:
        """鎸夐儴浠惰瀺鍚堣鍒欑粨鏋?""
        component_scores = {}
        
        # 鎸夐儴浠跺垎缁?
        rules_by_component = {}
        for result in rule_results:
            component = result.get('component', 'unknown')
            if component not in rules_by_component:
                rules_by_component[component] = []
            rules_by_component[component].append(result)
        
        # 瀵规瘡涓儴浠惰繘琛岃瀺鍚?
        for component, component_rules in rules_by_component.items():
            scores = [r['score'] for r in component_rules]
            weights = [r.get('weight', 1.0) for r in component_rules]
            confidences = [r.get('confidence', 1.0) for r in component_rules]
            
            fusion_func = self.fusion_strategies.get(strategy, self._exponential_weighted_fusion)
            component_scores[component] = fusion_func(scores, weights, confidences)
        
        return component_scores
```
